# title: מידע מרחבי — גרפים, מפות ולוויינים

ש: פֶּרֶק שֵׁנִי, יְחִידָה שְׁתַּיִם, מֵידָע מֶרְחָבִי. מֵאֵיפֹה מַתְחִילִים?

א: מִשְּׁנַת אֶלֶף שְׁבַע מֵאוֹת שְׁלוֹשִׁים וְחָמֵשׁ, בָּעִיר קֵנִיגְסְבֵּרְג.

ש: נִשְׁמָע רָחוֹק מִתּוֹבָלָה וְשִׁנּוּעַ.

א: זוֹ הַבְּעָיָה הָרִאשׁוֹנָה שֶׁנִּפְתְּרָה בְּתוֹרַת הַגְּרָפִים, וְכָל מָה שֶׁנַּעֲשָׂה בְּהֶמְשֵׁךְ הַקּוּרְס יוֹשֵׁב עָלֶיהָ. בָּעִיר עָבַר נָהָר וּבוֹ שִׁבְעָה גְּשָׁרִים. הַתּוֹשָׁבִים חִפְּשׂוּ מַסְלוּל מַעְגָּלִי שֶׁעוֹבֵר בְּכָל גֶּשֶׁר פַּעַם אַחַת בִּלְבַד, וְלֹא מָצְאוּ.

ש: וְאוֹייְלֶר הוֹכִיחַ שֶׁאִי אֶפְשָׁר.

א: וְזֶה הַחֵלֶק הַיָּפֶה. הוּא לֹא נִסָּה עוֹד מַסְלוּלִים. הוּא תִּרְגֵּם אֶת הַמְּצִיאוּת לְגָרֵף. כָּל יַבָּשָׁה הָפְכָה לַקּוֹדְקוֹד, כָּל גֶּשֶׁר לְקֶשֶׁת, וְאָז שָׁאַל מָה הַתְּנַאי הַכְּלָלִי.

ש: וּמָה הַתְּנַאי?

א: מִסְתַּכְּלִים עַל הַדַּרְגָּה שֶׁל כָּל קוֹדְקוֹד, כְּלוֹמַר כַּמָּה קְשָׁתוֹת מְחֻבָּרוֹת אֵלָיו. אִם כָּל הַקּוֹדְקוֹדִים בְּדַרְגָּה זוּגִית, קַיָּם מַסְלוּל מַעְגָּלִי, מַתְחִילִים וּמְסַיְּמִים בְּאוֹתָהּ נְקֻדָּה. אִם יֵשׁ בְּדִיּוּק שְׁנֵי קוֹדְקוֹדִים בְּדַרְגָּה אִי זוּגִית, יֵשׁ מַסְלוּל, אֲבָל חַיָּבִים לְהַתְחִיל בְּאֶחָד מֵהֶם וּלְסַיֵּם בַּשֵּׁנִי.

ש: וּבְקֵנִיגְסְבֵּרְג?

א: הָיוּ קוֹדְקוֹדִים בְּדַרְגָּה שָׁלוֹשׁ, שָׁלוֹשׁ וְחָמֵשׁ. יוֹתֵר מִשְּׁנַיִם אִי זוּגִיִּים, וְלָכֵן אֵין פִּתְרוֹן.

ש: יֵשׁ אֶת דֻּגְמַת צִיּוּר הַבַּיִת.

א: הָרִבּוּעַ עִם הָאִיקְס וְהַגַּג הַמְּשֻׁלָּשׁ, לְצַיֵּר בְּלִי לְהָרִים עִפָּרוֹן. שְׁתֵּי הַפִּנּוֹת הַתַּחְתּוֹנוֹת בְּדַרְגָּה שָׁלוֹשׁ, כְּלוֹמַר אִי זוּגִית, וְכָל הַשְּׁאָר זוּגִיּוּת. בְּדִיּוּק שְׁנַיִם אִי זוּגִיִּים, אָז אֶפְשָׁר. אֲבָל חַיָּבִים לְהַתְחִיל בְּאַחַת הַפִּנּוֹת הַתַּחְתּוֹנוֹת וּלְסַיֵּם בִּשְׁנִיָּה.

ש: וְאִם מַתְחִילִים בְּרֹאשׁ הַגַּג?

א: נִתְקָעִים. וְזוֹ בְּדִיּוּק הַנְּקֻדָּה: הַשְּׁאֵלָה הִיא לֹא רַק אִם אֶפְשָׁר, אֶלָּא גַּם מֵאֵיפֹה מַתְחִילִים.

ש: נַעֲבֹר לְגִ'י אֵיי אֵס.

א: מַעֲרֶכֶת מֵידָע גֵּאוֹגְרָפִית, בְּעִבְרִית מָמָ"ג. מַעֲרֶכֶת מְמֻחְשֶׁבֶת לִקְלִיטָה, אִחְסוּן, עִדְכּוּן, תִּשְׁאוּל וְהַצָּגָה שֶׁל נְתוּנִים גֵּאוֹגְרָפִיִּים, שֶׁמְּאַפְשֶׁרֶת גַּם לְנַתֵּחַ אוֹתָם וְלִיצֹר מֵידָע חָדָשׁ.

ש: וּמָה הַיִּתְרוֹן הַגָּדוֹל?

א: שֶׁהַנְּתוּנִים נִשְׁמָרִים עִם יִחוּס גֵּאוֹגְרָפִי. כְּלוֹמַר אֶפְשָׁר לְהַצִּיג אוֹתָם עַל מַפָּה, וְלֹא רַק בַּטַּבְלָה. וְזֶה מְשַׁנֶּה לְגַמְרֵי אֶת סוּג הַשְּׁאֵלוֹת שֶׁאֶפְשָׁר לִשְׁאֹל.

ש: חֲמֵשֶׁת מַרְכִּיבֵי גִ'י אֵיי אֵס. שְׁאֵלַת מִבְחָן קְלָאסִית.

א: נְתוּנִים, חֻמְרָה, תוֹכְנָה, פְּרוֹצֵדוּרוֹת, וְכוֹחַ אָדָם.

ש: וּפֹה יֵשׁ מַלְכֹּדֶת.

א: מַלְכֹּדֶת אֲמִתִּית. אַל תִּתְבַּלְבְּלוּ עִם שְׁלַבֵּי הַתַּהֲלִיךְ, שֶׁהֵם: אִסּוּף, אִחְסוּן, תִּשְׁאוּל, נִתּוּחַ, תְּצוּגָה, פֶּלֶט. חֲמֵשֶׁת הַמַּרְכִּיבִים זֶה מִמָּה הַמַּעֲרֶכֶת בְּנוּיָה. הַשְּׁלַבִּים זֶה מָה עוֹשִׂים אִתָּהּ.

ש: עַכְשָׁו, רַאסְטֶר מוּל וֶקְטוֹר. הַהַבְחָנָה שֶׁהֲכִי חוֹזֶרֶת.

א: רַאסְטֶר זוֹ תְּמוּנָה שֶׁמְּפֹרֶקֶת לַתָּאִים, כְּלוֹמַר פִּיקְסֵלִים. הָרֵזוֹלוּצְיָה נִקְבַּעַת לְפִי גֹּדֶל הַתָּא. כְּכָל שֶׁהִתְאִים קְטַנִּים יוֹתֵר, יֵשׁ יוֹתֵר פֵּרוּט.

ש: וְלָמָּה זֶה מַתְאִים?

א: תַּצְלוּמֵי אֲוִיר, תַּצְלוּמֵי לַוְיָן, מַפּוֹת סְרוּקוֹת. הַיִּתְרוֹן: מָהִיר לִקְלִיטָה, פָּשׁוּט סוֹרְקִים. הַחִסָּרוֹן: הַתְּמוּנָה קְבוּעָה, קָשֶׁה לְשַׁנּוֹת פֵּרוּט אוֹ לְהוֹסִיף נְתוּנִים.

ש: וֶקְטוֹר?

א: יִצּוּג מִסְפָּרִי בַּקּוֹאוֹרְדִּינָטוֹת. שָׁלוֹשׁ יֵשׁוּיוֹת בְּסִיסִיּוֹת. נְקֻדּוֹת, כְּמוֹ עַמּוּדֵי תְּאוּרָה, אַנְטֵנוֹת וּבְאֵרוֹת. קַוִּים, כְּמוֹ כְּבִישִׁים, נְהָרוֹת, פַּסֵּי רַכֶּבֶת וְצִנּוֹרוֹת. וּפּוֹלִיגוֹנִים, כְּלוֹמַר שְׁטָחִים, כְּמוֹ מִבְנִים, מִגְרָשִׁים וְיִעוּדִי קַרְקַע.

ש: וְהַיִּתְרוֹן?

א: שֶׁאֶפְשָׁר לְבַצֵּעַ עָלָיו חִשּׁוּבִים וּמָנִיפּוּלַצְיוֹת, וְהַתְּצוּגָה דִּינָמִית. אֶפְשָׁר לְהִתְקָרֵב וּלְהִתְרַחֵק בְּלִי לְאַבֵּד דִּיּוּק.

ש: אָז אִם בַּשְּׁאֵלָה מוֹפִיעַ תַּצְלוּם אֲוִיר,

א: זֶה רַאסְטֶר. וְאִם מוֹפִיעַ פּוֹלִיגוֹנִים עִם קוֹאוֹרְדִּינָטוֹת, זֶה וֶקְטוֹר. זֶה כִּמְעַט תָּמִיד הַהֶבְדֵּל שֶׁהַשְּׁאֵלָה בּוֹדֶקֶת.

ש: שְׁלֹשֶׁת מַרְכִּיבֵי הַמֵּידָע הַגֵּאוֹגְרָפִי?

א: גֵּאוֹמֶטְרִיָּה, כְּלוֹמַר הַמִּקּוּם וְהַצּוּרָה בַּמֶּרְחָב. מְאַפְיְנִים, שֶׁזֶּה הַנְּתוּנִים בַּטַּבְלָה, כְּמוֹ שֵׁם רְחוֹב אוֹ מִסְפַּר נְתִיבִים. וְחֻקִּים אוֹ הִתְנַהֲגוּת, שֶׁזֶּה יְחָסִים מֶרְחָבִיִּים וּכְלָלֵי הִגָּיוֹן, לְמָשָׁל שֶׁרְחוֹב מְקוֹמִי וּכְבִישׁ מָהִיר לֹא אֲמוּרִים לְהִצְטַלֵּב סְתָם.

ש: מָה זֶה עִגּוּן תְּמוּנָה?

א: תְּמוּנוֹת רַאסְטֶר בְּדֶרֶךְ כְּלָל לֹא מְכִילוֹת מֵידָע גֵּאוֹגְרָפִי, וְלִפְעָמִים גַּם צֻלְּמוּ בְּזָוִית לֹא מַתְאִימָה. עִגּוּן תְּמוּנָה הוּא לַסּוֹבֵב, לְהַגְדִּיל וּלְהַתְאִים אֶת הַתְּמוּנָה לְמַעֲרֶכֶת הַקּוֹאוֹרְדִּינָטוֹת שֶׁל הָעוֹלָם הָאֲמִתִּי.

ש: נַעֲבֹר לַנִּתּוּחַ. יֵשׁ שְׁלוֹשָׁה סוּגִים.

א: קִרְבָה, שְׁכָבוֹת עַל, וְרֶשֶׁת. וְאֶת שְׁלָשְׁתָּם צָרִיךְ לָדַעַת לְהַבְדִּיל.

ש: קִרְבָה?

א: מֶרְחָק מִמְּאַפְיֵן. אֵילוּ חֲלָקוֹת נִמְצָאוֹת בַּטְּוָח חֲמִשִּׁים מֶטֶר מֵהַכְּבִישׁ? אֵילוּ אֲזוֹרִים בְּסַכָּנַת שִׁטָּפוֹן לְפִי קִרְבָה לַנְּהָרוֹת?

ש: שְׁכָבוֹת עַל?

א: הַצְלָבָה שֶׁל כַּמָּה שְׁכָבוֹת בְּאוֹתוֹ מֶרְחָב. לְמָשָׁל: כָּל הַבְּאֵרוֹת מִסּוּג מְסֻיָּם, שֶׁנִּמְצָאוֹת בְּקַרְקַע חוֹלִית, וּבַעֲלוֹת מִבְנֶה מְסֻיָּם. כָּל תְּנַאי הוּא שִׁכְבָה אַחֶרֶת.

ש: וְרֶשֶׁת?

א: נִתּוּחַ לְאֹרֶךְ רֶשֶׁת קַוִּית. הַמַּסְלוּל הַקָּצָר מִבֵּית חוֹלִים לְזִירַת תְּאוּנָה, תּוֹךְ הִתְחַשְּׁבוּת בְּגֹבַהּ גְּשָׁרִים וּבְמִשְׁקָל מֻתָּר.

ש: אָז מִלַּת הַמַּפְתֵּחַ בַּשְּׁאֵלָה תִּהְיֶה מֶרְחָק, הַצְלָבָה, אוֹ מַסְלוּל.

א: בְּדִיּוּק כָּךְ. זֶה הַטְּרִיק לְזַהוֹת בַּמִּבְחָן.

ש: מָה עִם קְנֵה מִדָּה? זֶה מְבַלְבֵּל.

א: מְבַלְבֵּל, כִּי זֶה הָפוּךְ מֵהָאִינְטוּאִיצְיָה. מִסְפָּר קָטָן בִּמְכַנֶּה זֶה קְנֵה מִדָּה גָּדוֹל, וְזֶה יוֹתֵר פֵּרוּט. בְּאֶחָד לַחֲמֵשׁ מֵאוֹת, עִיר תּוֹפִיעַ כְּפּוֹלִיגוֹן. בְּאֶחָד לְמָאתַיִם חֲמִשִּׁים אֶלֶף, שֶׁזֶּה קְנֵה מִדָּה קָטָן, אוֹתָהּ עִיר תּוֹפִיעַ כַּנְּקֻדָּה.

ש: אֲבָל הַמֵּידָע עַצְמוֹ לֹא הִשְׁתַּנָּה.

א: נָכוֹן, הַכֹּל נִמְצָא בַּמַּעֲרֶכֶת. זֶה רַק עִנְיָן שֶׁל תְּצוּגָה.

ש: נַעֲבֹר לְגִ'י פִּי אֵס.

א: הַמַּעֲרֶכֶת הָאָמֵרִיקָאִית. עֶשְׂרִים וְאַרְבָּעָה לַוְיָנִים, שִׁשָּׁה מַסְלוּלִים סְבִיב כַּדּוּר הָאָרֶץ, אַרְבָּעָה לַוְיָנִים בְּכָל מַסְלוּל, בְּגֹבַהּ שֶׁל כְּעֶשְׂרִים אֶלֶף וּמָאתַיִם קִילוֹמֶטֶר. הוּקְמָה עַל יְדֵי מִשְׂרַד הַהֲגָנָה הָאָמֵרִיקָאִי.

ש: וְאֵיךְ הִיא עוֹבֶדֶת?

א: כָּל לַוְיָן נוֹשֵׂא שָׁעוֹן אָטוֹמִי מְדֻיָּק וּמַשְׁדֵּר אוֹת שֶׁכּוֹלֵל אֶת הַשָּׁעָה וְאֶת מִקּוּמוֹ. הַמִּקְלָט מְחַשֵּׁב אֶת הַמֶּרְחָק מֵהַלַּוְיָן לְפִי הַזְּמַן שֶׁלָּקַח לְאוֹת לְהַגִּיעַ.

ש: וּפֹה מַגִּיעָה הַשְּׁאֵלָה הַקְּלָאסִית: לָמָּה צָרִיךְ אַרְבָּעָה לַוְיָנִים?

א: כִּי יֵשׁ אַרְבָּעָה נֶעֱלָמִים. שָׁלוֹשׁ קוֹאוֹרְדִּינָטוֹת מֶרְחָב, אֹרֶךְ, רֹחַב וְגֹבַהּ, וּבְנוֹסָף הַזְּמַן.

ש: לָמָּה הַזְּמַן נֶעֱלַם? הֲרֵי לַשָּׁעוֹן הָאָטוֹמִי יֵשׁ זְמַן מְדֻיָּק.

א: לַלַּוְיָן יֵשׁ. לַמִּקְלָט שֶׁלָּכֶם, לֹא. בַּטֵּלֵפוֹן אֵין שָׁעוֹן אָטוֹמִי, וְלָכֵן יֵשׁ סְטִיָּה בִּשְׁעוֹן הַמִּקְלָט, וְהִיא נֶעְלָם רְבִיעִי. אַרְבָּעָה נֶעֱלָמִים דּוֹרְשִׁים אַרְבַּע מִשְׁוָאוֹת, וְלָכֵן אַרְבָּעָה לַוְיָנִים.

ש: וְזֶה נִקְרָא טְרִיאֶנְגּוֹלַצְיָה.

א: חִשּׁוּב נְקֻדָּה בַּמֶּרְחָב עַל יְדֵי הַצְלָבַת מֶרְחַקִּים מִכַּמָּה מְקוֹרוֹת.

ש: יֵשׁ מַעֲרָכוֹת נוֹסָפוֹת?

א: גָּלִילֵאוֹ בְּאֵירוֹפָּה, גְלוֹנַאס בְּרוּסְיָה, וּבְיָיָיָדוֹ בְּסִין.

ש: מָה זֶה אֵיי גִ'י פִּי אֵס?

א: הַבְּעָיָה עִם גִ'י פִּי אֵס הִיא שֶׁהוּא צָרִיךְ לִטְעֹן אֶת כָּל נְתוּנֵי מִקּוּמֵי הַלַּוְיָנִים לִפְנֵי שֶׁהוּא מַתְחִיל לַעֲבֹד, וְזֶה אִטִּי, בִּמְיֻחָד כְּשֶׁהָאוֹת חַלָּשׁ. אֵיי גִ'י פִּי אֵס, כְּלוֹמַר גִ'י פִּי אֵס נֶעֱזָר, מוֹרִיד אֶת הַנְּתוּנִים הָאֵלֶּה מִשָּׂרַת אִינְטֶרְנֵט בִּמְקוֹם מֵהַלַּוְיָן.

ש: וְזֶה מֵאִיץ אֶת הַנְּעִילָה הָרִאשׁוֹנָה.

א: שֶׁנִּקְרֵאת טִי טִי אֵף אֵף, זְמַן לִנְעִילָה רִאשׁוֹנָה. מַתְאִים לְמַכְשִׁירִים סֵלוֹלָרִיִּים שֶׁיְּכוֹלִים לְהִתְחַבֵּר לָאִינְטֶרְנֵט, וּבְעִקָּר בֶּעָרִים.

ש: וְעַכְשָׁו אֵיי וִי אֵל, מֻשָּׂג מֶרְכָּזִי לַקּוּרְס.

א: הַמִּשְׁוָאָה פְּשׁוּטָה. אֵיי וִי אֵל שָׁוֶה גִ'י פִּי אֵס וְעוֹד גִ'י אֵיי אֵס. אִתּוּר אוֹטוֹמָטִי שֶׁל כְּלֵי רֶכֶב. קֻפְסָה מִתְחַבֶּרֶת לְרֶכֶב, קוֹלֶטֶת נְתוּנֵי מִקּוּם וּמַעֲבִירָה לְמוֹקֵד הַחֶבְרָה, שֶׁמַּצִּיג אוֹתָם עַל מַפָּה.

ש: וּמָה זֶה נוֹתֵן בְּפֹעַל?

א: אַרְבָּעָה דְּבָרִים. הַצָּגַת כָּל הַצִּי עַל מַפָּה. שִׁלּוּחַ דִּינָמִי, כְּלוֹמַר לִבְדֹּק מִי פָּנוּי וְקָרוֹב וְלִשְׁלֹחַ אוֹתוֹ, לְמָשָׁל אַמְבּוּלַנְס. אֲזוֹרִים מְגֻדָּרִים, גִ'יאוֹפֶנְסִינְג, שֶׁזֶּה הַתְרָאָה כְּשֶׁרָכַב נִכְנַס אוֹ יוֹצֵא מֵאֵזוֹר מֻגְדָּר. וְתִקְשֹׁרֶת עִם כְּלֵי הָרֶכֶב, כּוֹלֵל מַעֲקָב מְלַאי בְּשִׁלּוּב אָר אֵף אֵיי דִּי.

ש: אָז אִם בִּשְׁאֵלָה מוֹפִיעַ לִרְאוֹת אֵיפֹה הַמַּשָּׂאִיּוֹת וּלְקַבֵּל הַתְרָאָה עַל יְצִיאָה מֵאֵזוֹר,

א: זֶה אֵיי וִי אֵל. וְזוֹ הַתְּשׁוּבָה, לֹא אָר אַף אֵיי דִּי לְבַדּוֹ.

ש: נְסַיֵּם בְּאִכּוּן בְּתוֹךְ מִבְנִים.

א: בְּתוֹךְ מִבְנִים אוֹתוֹת הַגִּ'י פִּי אֵס חַלָּשִׁים אוֹ לֹא זְמִינִים בִּכְלָל, וְלִפְעָמִים גַּם צָרִיךְ דִּיּוּק גָּבוֹהַּ יוֹתֵר. לָכֵן מִשְׁתַּמְּשִׁים בְּחַיְשָׁנִים שֶׁכְּבָר יֵשׁ בַּטֵּלֵפוֹן: מַצְפֵּן, מַד תְּאוּצָה, גִ'ירוֹסְקופּ, מַד צְעָדִים וּמַד גֹּבַהּ.

ש: וּבְנוֹסָף?

א: וַויי פַיי, שֶׁבּוֹ הַטֵּלֵפוֹן קוֹלֵט נְקֻדּוֹת גִּישָׁה מְמֻפּוֹת וּלְפִי עוֹצְמַת הָאוֹת עוֹשֶׂה מֵעֵין טְרִיאַנְגּוֹלַצְיָה. וּבַלּוּטוֹת', וּבְעִקָּר בִּי אֶל אֵיי, שֶׁהַטְּוָח שֶׁלּוֹ כַּחֲמִשִּׁים מֶטֶר.

ש: וּמָה הַיִּתְרוֹן שֶׁל בַּלּוּטוֹת' עַל וַויי פַיי?

א: צְרִיכַת אֵנֶרְגִּיָּה נְמוּכָה בְּהַרְבֵּה. וְזֶה מוֹבִיל לְסֵדֶר שֶׁכְּדַאי לִזְכֹּר, מֵהַחֶסְכוֹנִי לְבַזְבְּזָנִי: בִּי אֶל אֵיי, שֶׁנּוֹתֵן בְּעֵרֶךְ מֵאָה יְמֵי סוֹלְלָה. אַחֲרָיו בַּלּוּטוֹת' שְׁתַּיִם נְקֻדָּה אַחַת. אַחַר כָּךְ מַעֲרֶכֶת נִוּוּט אִינֶרְצִיָּאלִית. אַחַר כָּךְ וַויי פַיי. וּבַסּוֹף, גִ'י פִּי אֵס, שֶׁנּוֹתֵן שָׁעָה עַד שְׁעָתַיִם וָחֵצִי בִּלְבַד.

ש: אָז גִ'י פִּי אֵס הֲכִי מְדֻיָּק בַּחוּץ, וַהֲכִי יָקָר בָּאֵנֶרְגִּיָּה.

א: בְּדִיּוּק. וְזֶה הַטְּרֵיְד אוֹף.

ש: שְׁאֵלוֹת לַדֶּרֶךְ. רִאשׁוֹנָה: מָתַי קַיָּם מַסְלוּל אוֹיְלֶר, וּמָתַי הוּא מַעְגָּלִי?

~4

א: כְּשֶׁיֵּשׁ בְּדִיּוּק שְׁנֵי קוֹדְקוֹדִים בְּדַרְגָּה אִי זוּגִית, קַיָּם מַסְלוּל, וּמַתְחִילִים בְּאֶחָד וּמְסַיְּמִים בַּשֵּׁנִי. כְּשֶׁכָּל הַקּוֹדְקוֹדִים בְּדַרְגָּה זוּגִית, הַמַּסְלוּל מַעְגָּלִי.

ש: שְׁנִיָּה: תַּצְלוּם אֲוִיר מְחֻלָּק לְרֶשֶׁת תָּאִים. רַאסְטֶר אוֹ וֶקְטוֹר?

~4

א: רַאסְטֶר. תָּאִים זֶה רַאסְטֶר, קוֹאוֹרְדִּינָטוֹת זֶה וֶקְטוֹר.

ש: שְׁלִישִׁית: מָה חֲמֵשֶׁת מַרְכִּיבֵי גִ'י אֵיי אֵס, וּבְמָה הֵם שׁוֹנִים מִשְׁלַבֵּי הַתַּהֲלִיךְ?

~4

א: הַמַּרְכִּיבִים הֵם נְתוּנִים, חֻמְרָה, תוֹכְנָה, פְּרוֹצֵדוּרוֹת וְכוֹחַ אָדָם. הַשְּׁלַבִּים הֵם אִסּוּף, אִחְסוּן, תִּשְׁאוּל, נִתּוּחַ, תְּצוּגָה וּפֶלֶט. הַמַּרְכִּיבִים זֶה מִמָּה הַמַּעֲרֶכֶת בְּנוּיָה, הַשְּׁלַבִּים זֶה מָה עוֹשִׂים אִתָּהּ.

ש: רְבִיעִית: לָמָּה נִדְרָשִׁים אַרְבָּעָה לַוְיָנִים וְלֹא שְׁלוֹשָׁה?

~4

א: כִּי יֵשׁ אַרְבָּעָה נֶעֱלָמִים, לֹא שְׁלוֹשָׁה. אֹרֶךְ, רֹחַב, גֹּבַהּ, וּבְנוֹסָף הַזְּמַן, בִּגְלַל סְטִיַּת הַשָּׁעוֹן שֶׁל הַמִּקְלָט.

ש: וְאַחֲרוֹנָה: חֶבְרָה רוֹצָה לִרְאוֹת אֶת הַמַּשָּׂאִיּוֹת עַל מַפָּה וּלְקַבֵּל הַתְרָאָה כְּשֶׁרָכַב יוֹצֵא מֵאֵזוֹר. בְּאֵיזוֹ טֶכְנוֹלוֹגְיָה מְדֻבָּר?

~4

א: אֵיי וִי אֵל, שֶׁהוּא גִ'י פִּי אֵס וְעוֹד גִ'י אֵיי אֵס. הַהַתְרָאָה עַל יְצִיאָה מֵאֵזוֹר נִקְרֵאת גִ'יאוֹפֶנְסִינְג.

ש: בַּפֶּרֶק הַבָּא, זְרִימָה בָּרְשָׁתוֹת. שָׁם מַתְחִילוֹת הַבְּעָיוֹת הַמְּעַנְיְנוֹת בֶּאֱמֶת.
